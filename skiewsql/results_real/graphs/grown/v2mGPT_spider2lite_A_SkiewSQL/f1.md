_Task: For all races that have pit stop data available, count how many overtakes of each overtake-type occurred, where an overtake is defined as any instance in which a given driver was not classified _

```sql
SELECT
  lp_prev.race_id,
  lp_prev.driver_id AS subject_driver_id,
  lp_prev.lap AS lap_prev,
  lp_curr.lap AS lap_curr,
  lp_prev.position AS position_prev,
  lp_curr.position AS position_curr,
  lp_prev.lap_type AS lap_type_prev,
  lp_curr.lap_type AS lap_type_curr,
  rs.race_id IS NOT NULL AS has_pitstop_record,
  p_prev.stop AS pit_stop_prev_lap,
  p_curr.stop AS pit_stop_curr_lap,
  r.retirement_type AS overtaken_retirement_type,
  r.status_id AS overtaken_retirement_status_id,
  r.lap AS overtaken_retirement_lap,
  -- keys to join later: potential overtaken driver id will be available via joins in downstream query
  lp_prev.rowid AS lp_prev_rowid,
  lp_curr.rowid AS lp_curr_rowid
FROM
  lap_positions lp_prev
JOIN
  lap_positions lp_curr
  ON lp_prev.race_id = lp_curr.race_id
  AND lp_prev.driver_id = lp_curr.driver_id
  AND lp_curr.lap = lp_prev.lap + 1
LEFT JOIN
  pit_stops p_prev
  ON p_prev.race_id = lp_prev.race_id
  AND p_prev.driver_id = lp_prev.driver_id
  AND p_prev.lap = lp_prev.lap
LEFT JOIN
  pit_stops p_curr
  ON p_curr.race_id = lp_curr.race_id
  AND p_curr.driver_id = lp_curr.driver_id
  AND p_curr.lap = lp_curr.lap
LEFT JOIN
  retirements r
  ON r.race_id = lp_curr.race_id
  AND r.driver_id = lp_curr.driver_id
LEFT JOIN
  (SELECT DISTINCT race_id FROM pit_stops) rs
  ON rs.race_id = lp_prev.race_id
WHERE
  rs.race_id IS NOT NULL
  AND lp_prev.position IS NOT NULL
  AND lp_curr.position IS NOT NULL
  -- driver was NOT behind on previous lap but IS behind on current lap
  AND (lp_prev.position IS NULL OR lp_prev.position <= lp_curr.position - 1) = FALSE
  AND lp_prev.position < lp_curr.position
```
