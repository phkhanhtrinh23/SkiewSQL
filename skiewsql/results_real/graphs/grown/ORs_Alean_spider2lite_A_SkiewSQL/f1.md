_Overtake events detected by comparing each driver's position relative to other cars across consecutive laps, classified by associated event (retirement, pit entry, pit exit), for races with pit stop data._

```sql
SELECT
    o.race_id,
    CASE
        WHEN r.race_id IS NOT NULL THEN 'retirement'
        WHEN ps_o.race_id IS NOT NULL THEN 'pit_exit'
        WHEN ps_d.race_id IS NOT NULL THEN 'pit_entry'
        ELSE 'on_track'
    END AS overtake_type,
    o.overtaking_driver_id,
    o.overtaken_driver_id,
    o.lap_number
FROM (
    SELECT
        c1.race_id,
        c1.driver_id AS overtaking_driver_id,
        c2.driver_id AS overtaken_driver_id,
        c1.lap AS lap_number
    FROM lap_positions c1
    JOIN lap_positions c2
        ON c1.race_id = c2.race_id
       AND c1.lap = c2.lap
       AND c1.driver_id <> c2.driver_id
       AND c1.lap_type = 'Race'
       AND c2.lap_type = 'Race'
    JOIN lap_positions p1
        ON p1.race_id = c1.race_id
       AND p1.driver_id = c1.driver_id
       AND p1.lap = c1.lap - 1
       AND p1.lap_type = 'Race'
    JOIN lap_positions p2
        ON p2.race_id = c2.race_id
       AND p2.driver_id = c2.driver_id
       AND p2.lap = c2.lap - 1
       AND p2.lap_type = 'Race'
    WHERE c1.race_id IN (SELECT race_id FROM races_with_pit_data)
      AND p1.position > p2.position
      AND c1.position < c2.position
) o
LEFT JOIN (
    SELECT DISTINCT race_id, driver_id, CAST(lap AS INT) AS lap
    FROM retirements
) r
    ON r.race_id = o.race_id
   AND r.driver_id = o.overtaken_driver_id
   AND r.lap = o.lap_number
LEFT JOIN (
    SELECT DISTINCT race_id, driver_id, lap
    FROM pit_stops
) ps_o
    ON ps_o.race_id = o.race_id
   AND ps_o.driver_id = o.overtaking_driver_id
   AND ps_o.lap = o.lap_number
LEFT JOIN (
    SELECT DISTINCT race_id, driver_id, lap
    FROM pit_stops
) ps_d
    ON ps_d.race_id = o.race_id
   AND ps_d.driver_id = o.overtaken_driver_id
   AND ps_d.lap = o.lap_number
```
